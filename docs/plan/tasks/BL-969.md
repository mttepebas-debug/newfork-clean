```sql
-- BL-969 live measurement, read only (run before the migration, Mert's "evet" first).
-- 1. Rows the letter fix would change, deleted catalog rows counted on their own.
with letters as (
  select 'workspaces.name' as col, count(*) filter (where name ~ '[şŞğĞıİ]') as bad from public.workspaces
  union all select 'locations (live)', count(*) filter (where deleted_at is null and (name ~ '[şŞğĞıİ]' or street_address ~ '[şŞğĞıİ]' or city ~ '[şŞğĞıİ]' or region ~ '[şŞğĞıİ]')) from public.locations
  union all select 'locations (deleted)', count(*) filter (where deleted_at is not null and (name ~ '[şŞğĞıİ]' or street_address ~ '[şŞğĞıİ]' or city ~ '[şŞğĞıİ]' or region ~ '[şŞğĞıİ]')) from public.locations
  union all select 'departments (live)', count(*) filter (where deleted_at is null and name ~ '[şŞğĞıİ]') from public.departments
  union all select 'departments (deleted)', count(*) filter (where deleted_at is not null and name ~ '[şŞğĞıİ]') from public.departments
  union all select 'jobs (live)', count(*) filter (where deleted_at is null and name ~ '[şŞğĞıİ]') from public.jobs
  union all select 'jobs (deleted)', count(*) filter (where deleted_at is not null and name ~ '[şŞğĞıİ]') from public.jobs
  union all select 'employee_profiles names', count(*) filter (where first_name ~ '[şŞğĞıİ]' or middle_name ~ '[şŞğĞıİ]' or last_name ~ '[şŞğĞıİ]' or preferred_name ~ '[şŞğĞıİ]') from public.employee_profiles
  union all select 'private profile names/address/contact', count(*) filter (where legal_first_name ~ '[şŞğĞıİ]' or legal_middle_name ~ '[şŞğĞıİ]' or legal_last_name ~ '[şŞğĞıİ]' or preferred_name ~ '[şŞğĞıİ]' or street_address ~ '[şŞğĞıİ]' or city ~ '[şŞğĞıİ]' or region ~ '[şŞğĞıİ]' or emergency_contact_name ~ '[şŞğĞıİ]' or emergency_contact_relationship ~ '[şŞğĞıİ]') from public.employee_private_profiles
  union all select 'preferred name requests', count(*) filter (where requested_name ~ '[şŞğĞıİ]') from public.employee_preferred_name_requests
)
select 'letters' as kind, col, bad from letters
union all
-- 2. Phones by shape: E.164 US, US without +1 (fixed), anything else (stops if required).
select 'phone', col, count(*) from (
  select 'private.phone ' || case
      when phone ~ '^\+1[2-9][0-9]{9}$' then 'e164'
      when regexp_replace(phone, '\D', '', 'g') ~ '^1?[2-9][0-9]{9}$' and phone !~ '^\s*\+[02-9]' then 'us-raw'
      else 'other' end as col
  from public.employee_private_profiles where phone is not null
  union all
  select 'private.emergency ' || case
      when emergency_contact_phone ~ '^\+1[2-9][0-9]{9}$' then 'e164'
      when regexp_replace(emergency_contact_phone, '\D', '', 'g') ~ '^1?[2-9][0-9]{9}$' and emergency_contact_phone !~ '^\s*\+[02-9]' then 'us-raw'
      else 'other' end
  from public.employee_private_profiles where emergency_contact_phone is not null
  union all
  select 'account.phone_e164 ' || case when phone_e164 ~ '^\+1[2-9][0-9]{9}$' then 'e164' else 'other' end
  from public.user_profiles where phone_e164 is not null
  union all
  select 'account.old phone column (phone_e164 empty) ' || case
      when regexp_replace(phone, '\D', '', 'g') ~ '^1?[2-9][0-9]{9}$' and phone !~ '^\s*\+[02-9]' then 'us' else 'other' end
  from public.user_profiles where phone is not null and phone_e164 is null
) shapes group by col
union all
-- The case the migration's stop count leaves to this measure: an account number
-- that is not a US one, a US number in the old column, and a record holding a
-- different US number (the old number would reach the record).
select 'phone', 'account foreign + old US + record differs', count(*)
from public.user_profiles account
where account.phone_e164 is not null and account.phone_e164 !~ '^\+1[2-9][0-9]{9}$'
  and regexp_replace(account.phone, '\D', '', 'g') ~ '^1?[2-9][0-9]{9}$'
  and exists (
    select 1 from public.employee_profiles record
    join public.employee_private_profiles private_record on private_record.employee_id = record.id
    where record.user_id = account.user_id and private_record.phone is not null
      and right(regexp_replace(private_record.phone, '\D', '', 'g'), 10) <> right(regexp_replace(account.phone, '\D', '', 'g'), 10)
  )
union all
-- 3. The area codes the fix would give a +1 to, to read once by eye (a foreign
--    number typed without its country code reads as a US one).
select 'area code', area, count(*) from (
  select substr(regexp_replace(value, '\D', '', 'g'), case when regexp_replace(value, '\D', '', 'g') ~ '^1' and char_length(regexp_replace(value, '\D', '', 'g')) = 11 then 2 else 1 end, 3) as area
  from (
    select phone as value from public.employee_private_profiles where phone is not null and phone !~ '^\+1[2-9][0-9]{9}$'
    union all select emergency_contact_phone from public.employee_private_profiles where emergency_contact_phone is not null and emergency_contact_phone !~ '^\+1[2-9][0-9]{9}$'
  ) raw
) areas group by area
order by 1, 2;
```
