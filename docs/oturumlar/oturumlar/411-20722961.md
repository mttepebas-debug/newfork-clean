     # release <who> | status

     cmd = sys.argv[1]
     who = sys.argv[2] if len(sys.argv) > 2 else ''
     if cmd == 'take':
         deadline = time.time() + (float(sys.argv[3]) if len(sys.argv) > 3 else 600)
         while True:

                 os.write(fd, json.dumps({'who': who, 'at': time.time()}).encode()); os.close(fd)
